const {chromium}=require('../../frontend/node_modules/playwright');
const fs=require('node:fs'),path=require('node:path'),assert=require('node:assert/strict');
const base=process.env.HP_DEMO_PREVIEW||'http://127.0.0.1:18528';
const output=path.join(__dirname,process.env.HP_DEMO_SMOKE_RUN||'presentation-smoke');
(async()=>{fs.mkdirSync(output,{recursive:true});const browser=await chromium.launch({channel:'msedge',headless:true});
try{const context=await browser.newContext({viewport:{width:1440,height:1000}});const page=await context.newPage();const errors=[],apiRequests=[],checks=[];
page.on('pageerror',e=>errors.push(e.message));await context.route('**/api/**',r=>{apiRequests.push(r.request().url());return r.abort();});
await page.goto(base,{waitUntil:'networkidle'});await page.getByRole('heading',{level:1}).filter({hasText:'Cada forma de cuidar'}).waitFor();assert.equal(await page.locator('.review-journey').count(),6);await page.screenshot({path:path.join(output,'portada-desktop.png'),fullPage:true});checks.push('Portada: seis recorridos, sin login');
for(const name of ['Administración SaaS','Medicina humana','Odontología','Veterinaria']){await page.goto(base,{waitUntil:'networkidle'});await page.locator('.review-journey').filter({hasText:name}).click();await page.waitForLoadState('networkidle');assert.ok((await page.locator('body').innerText()).length>200);assert.equal((new URL(page.url())).origin,base);checks.push('Abre '+name);if(name==='Administración SaaS'){await page.getByRole('link',{name:'Clientes y tenants',exact:true}).click();await page.locator('.saas-tenant-card,.data-table').first().waitFor();checks.push('SaaS: clientes disponibles');await page.screenshot({path:path.join(output,'saas-clientes.png'),fullPage:true});}}
await page.goto(base,{waitUntil:'networkidle'});await page.setViewportSize({width:390,height:844});for(const theme of ['light','dark','navy']){await page.getByLabel('Tema de la presentación').selectOption(theme);assert.ok(await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth+1));await page.screenshot({path:path.join(output,'portada-movil-'+theme+'.png'),fullPage:true});}checks.push('Portada móvil390 en tres temas sin overflow');assert.deepEqual(apiRequests,[]);assert.deepEqual(errors,[]);const result={base,mode:'compiled standalone demo with synthetic fixtures',checks,apiRequests,errors,physicalAudioTested:false};fs.writeFileSync(path.join(output,'result.json'),JSON.stringify(result,null,2));console.log(JSON.stringify(result));await context.close();}finally{await browser.close();}})().catch(e=>{console.error(e);process.exitCode=1;});
