   const key=req.headers()['idempotency-key'];assert.equal(req.headers()['x-csrf-token'],'csrf-synthetic-only');assert.ok(key);

   const person=state.patients.find(p=>p.id===body.patientId),saved={...body,id:randomUUID(),patientName:person.name,identifier:person.identifier,professionalName:'Profesional QA',revision:1,collectionStatus:'PENDING',amountMinor:0,paymentMethod:null,collectionRevision:0};
   state.appointments.push(saved);state.bookingReceipts.set(key,{...saved});if(state.loseBooking){state.loseBooking=false;return route.abort('connectionfailed');}return send(saved);
  }
  const collection=/^\/api\/v1\/agenda\/citas\/([^/]+)\/cobro$/.exec(p);
  if(collection&&method==='PUT'){
   const a=state.appointments.find(a=>a.id===collection[1]);assert.equal(req.headers()['x-csrf-token'],'csrf-synthetic-only');
   if(state.collectionReceipts.has(body.requestId))return send(state.collectionReceipts.get(body.requestId));
   if(state.conflict)return send({detail:'Cobro actualizado por otra sesión de QA.'},409);
   Object.assign(a,{collectionStatus:body.status,amountMinor:body.amountMinor,paymentMethod:body.paymentMethod,collectionRevision:a.collectionRevision+1});
   const saved={appointmentId:a.id,status:a.collectionStatus,amountMinor:a.amountMinor,paymentMethod:a.paymentMethod,revision:a.collectionRevision};state.collectionReceipts.set(body.requestId,saved);
   if(state.loseCollection){state.loseCollection=false;return route.abort('connectionfailed');}return send(saved);
  }
  state.errors.push('Unexpected API '+method+' '+p);return send({detail:'Ruta no prevista en harness'},500);
 });
 return {page,state};
}
